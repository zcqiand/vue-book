<script setup lang="ts">
// 方式一：运行时声明（Vue 2 延续写法，不推荐新项目使用）
// 缺点：类型字符串容易写错，IDE 无法推导，ref 需手动断言类型
// defineProps({
//   title: { type: String, required: true },
//   count: { type: Number, default: 0 },
// })

// 方式二：基于类型的泛型声明（Vue 3.3+ 推荐写法）
interface Props {
  title: string
  count?: number
  tags?: string[]
  publishedAt?: Date
}

// withDefaults 处理可选属性的默认值
const props = withDefaults(defineProps<Props>(), {
  count: 0,
  tags: () => [],
  publishedAt: undefined,
})

// Vue 3.5 解构默认值语法（无需 withDefaults）
// const props = defineProps<Props & { count?: number; tags?: string[] }>()
// 直接在模板中使用 {{ title }}，默认值由解构提供
</script>

<template>
  <div class="book-card">
    <h2>{{ props.title }}</h2>
    <p>阅读量：{{ props.count.toLocaleString() }}</p>
    <div class="tags">
      <span v-for="tag in props.tags" :key="tag" class="tag">{{ tag }}</span>
    </div>
    <p v-if="props.publishedAt">
      发布时间：{{ props.publishedAt.toLocaleDateString() }}
    </p>
  </div>
</template>
