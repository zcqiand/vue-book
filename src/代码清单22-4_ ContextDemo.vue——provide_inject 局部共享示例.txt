<!-- src/components/ContextDemo.vue -->
<template>
  <div class="context-demo">
    <h3>provide/inject 适用场景：组件树局部上下文</h3>

    <!-- 表单上下文：多个子组件共享同一份表单数据 -->
    <FormContext>
      <FormField name="username" label="用户名" />
      <FormField name="email" label="邮箱" />
      <FormSubmit />
    </FormContext>

    <hr />

    <!-- 主题上下文：整棵子树共享同一主题 -->
    <ThemeContext>
      <HeaderSection />
      <ContentSection />
    </ThemeContext>
  </div>
</template>

<script setup lang="ts">
import { provide, inject, reactive, type InjectionKey } from 'vue'

// ============================================================
// 场景一：表单上下文（provide/inject 适合：组件树局部共享）
// ============================================================

interface FormContextValue {
  values: Record<string, string>
  errors: Record<string, string | null>
  setField: (name: string, value: string) => void
  setError: (name: string, error: string | null) => void
}

const formContextKey: InjectionKey<FormContextValue> = Symbol('form')

function provideFormContext(): void {
  const values = reactive<Record<string, string>>({})
  const errors = reactive<Record<string, string | null>>({})

  const setField = (name: string, value: string): void => {
    values[name] = value
    // 清除该字段的错误
    if (errors[name]) {
      errors[name] = null
    }
  }

  const setError = (name: string, error: string | null): void => {
    errors[name] = error
  }

  provide(formContextKey, { values, errors, setField, setError })
}

// FormContext 组件：提供表单上下文
const FormContext = {
  setup(_: unknown, { slots }: { slots: { default?: () => unknown } }) {
    provideFormContext()
    return () => slots.default?.()
  },
}

// FormField 组件：注入表单上下文
const FormField = {
  props: ['name', 'label'],
  setup(props: { name: string; label: string }, { slots }: { slots: { default?: () => unknown } }) {
    const form = inject(formContextKey)!

    return () => {
      const value = form.values[props.name] ?? ''
      const error = form.errors[props.name]
      return `
        <div class="form-field">
          <label>${props.label}</label>
          <input
            value="${value}"
            onInput="${(e: Event) => form.setField(props.name, (e.target as HTMLInputElement).value)}"
          />
          ${error ? `<span class="error">${error}</span>` : ''}
        </div>
      `
    }
  },
}

// FormSubmit 组件：注入表单上下文提交
const FormSubmit = {
  setup(_: unknown, { slots }: { slots: { default?: () => unknown } }) {
    const form = inject(formContextKey)!

    const handleSubmit = (): void => {
      console.log('表单值:', form.values)
    }

    return () => `
      <button onClick="${handleSubmit}">提交</button>
    `
  },
}

// ============================================================
// 场景二：主题上下文（provide/inject 适合：组件树局部共享）
// ============================================================

type Theme = 'light' | 'dark'

interface ThemeContextValue {
  theme: Theme
  toggleTheme: () => void
}

const themeContextKey: InjectionKey<ThemeContextValue> = Symbol('theme')

// ThemeContext 组件：提供主题上下文
const ThemeContext = {
  setup(_: unknown, { slots }: { slots: { default?: () => unknown } }) {
    const theme = reactive<{ value: Theme }>({ value: 'light' })

    const toggleTheme = (): void => {
      theme.value = theme.value === 'light' ? 'dark' : 'light'
    }

    provide(themeContextKey, { theme, toggleTheme })
    return () => slots.default?.()
  },
}

// HeaderSection 和 ContentSection：注入主题上下文
const HeaderSection = {
  setup(_: unknown, { slots }: { slots: { default?: () => unknown } }) {
    const ctx = inject(themeContextKey)!
    return () => `
      <header class="${ctx.theme.value}">
        主题：${ctx.theme.value} |
        <button onClick="${ctx.toggleTheme}">切换</button>
      </header>
    `
  },
}

const ContentSection = {
  setup(_: unknown, { slots }: { slots: { default?: () => unknown } }) {
    const ctx = inject(themeContextKey)!
    return () => `
      <main class="${ctx.theme.value}">
        内容区域，感知主题：${ctx.theme.value}
      </main>
    `
  },
}
</script>
