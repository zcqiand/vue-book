<!-- src/pages/RegisterForm.vue -->
<template>
  <div class="register-form">
    <h2>用户注册</h2>

    <form @submit.prevent="handleSubmit" class="form-container">
      <!-- 用户名 -->
      <div class="field-group">
        <label for="username">用户名</label>
        <input
          id="username"
          v-model="form.username"
          type="text"
          placeholder="请输入用户名"
          autocomplete="username"
        />
        <span v-if="errors.username" class="error-msg">{{ errors.username }}</span>
      </div>

      <!-- 邮箱 -->
      <div class="field-group">
        <label for="email">邮箱</label>
        <input
          id="email"
          v-model="form.email"
          type="email"
          placeholder="example@domain.com"
          autocomplete="email"
        />
        <span v-if="errors.email" class="error-msg">{{ errors.email }}</span>
      </div>

      <!-- 密码 -->
      <div class="field-group">
        <label for="password">密码</label>
        <input
          id="password"
          v-model="form.password"
          type="password"
          placeholder="至少 8 位"
          autocomplete="new-password"
        />
        <span v-if="errors.password" class="error-msg">{{ errors.password }}</span>
      </div>

      <!-- 确认密码 -->
      <div class="field-group">
        <label for="confirmPassword">确认密码</label>
        <input
          id="confirmPassword"
          v-model="form.confirmPassword"
          type="password"
          placeholder="再次输入密码"
          autocomplete="new-password"
        />
        <span v-if="errors.confirmPassword" class="error-msg">{{ errors.confirmPassword }}</span>
      </div>

      <button type="submit" class="btn-submit">注册</button>
    </form>
  </div>
</template>

<script setup lang="ts">
import { reactive } from 'vue'
import { useValidation, type validatorFn } from '../composables/useValidation'

// ---- 验证规则（validatorFn 数组）----

const required: validatorFn = (value: unknown): string | null => {
  if (value === null || value === undefined || (typeof value === 'string' && value.trim() === '')) {
    return '此字段为必填项'
  }
  return null
}

const minLength = (min: number): validatorFn => (value: unknown): string | null => {
  const str = value as string
  if (str.length < min) {
    return `至少需要 ${min} 个字符`
  }
  return null
}

const emailFormat: validatorFn = (value: unknown): string | null => {
  const str = value as string
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/
  if (!emailRegex.test(str)) {
    return '请输入有效的邮箱格式'
  }
  return null
}

const minLen8: validatorFn = (value: unknown): string | null => {
  const str = value as string
  if (str.length < 8) {
    return '密码至少需要 8 位'
  }
  return null
}

// ---- 表单数据 ----

const form = reactive({
  username: '',
  email: '',
  password: '',
  confirmPassword: '',
})

// ---- 调用 useValidation ----

// passwordMatch 必须放在 form 声明之后，因为 validator 函数通过闭包引用 form
const passwordMatch: validatorFn = (value: unknown): string | null => {
  if (value !== form.password) {
    return '两次输入的密码不一致'
  }
  return null
}

const { errors, validate, hasErrors } = useValidation(form, {
  username: [required, minLength(4)],
  email: [required, emailFormat],
  password: [required, minLen8],
  confirmPassword: [required, passwordMatch],
})

// ---- 提交处理 ----

function handleSubmit(): void {
  if (validate()) {
    console.log('表单提交成功:', form)
    alert('注册成功！')
  } else {
    console.log('表单有错误，阻止提交')
  }
}
</script>
