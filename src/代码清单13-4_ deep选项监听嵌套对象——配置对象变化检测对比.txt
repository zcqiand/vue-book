<!-- src/components/AppConfig.vue -->
<script setup lang="ts">
import { reactive, watch } from 'vue'

interface AppConfig {
  theme: string
  language: string
  notifications: {
    email: boolean
    push: boolean
    digest: boolean
  }
  timeout: number
}

const config = reactive<AppConfig>({
  theme: 'light',
  language: 'zh-CN',
  notifications: {
    email: true,
    push: false,
    digest: true,
  },
  timeout: 5000,
})

// 不带 deep：只检测 config 引用本身是否被替换
// config.notifications = { email: false, push: false, digest: false } 时触发
// 但 config.notifications.email = false 时不触发
watch(config, (newVal: AppConfig) => {
  console.log('[deep:false] config 整体变化:', newVal)
})

// 带 deep：递归检测嵌套属性任意层级的变化
// config.notifications.email 从 true 改成 false 时立即触发
watch(
  config,
  (newVal: AppConfig, oldVal: AppConfig) => {
    console.log('[deep:true] config 任意属性变化')
    console.log('  theme:', oldVal.theme, '→', newVal.theme)
    console.log('  notifications.email:', oldVal.notifications.email, '→', newVal.notifications.email)
  },
  { deep: true }
)

function toggleEmail(): void {
  // deep: true 时，每次切换都会触发 watch 回调
  // deep: false 时，此操作不会触发（因为 config 引用未变）
  config.notifications.email = !config.notifications.email
}

function changeTheme(): void {
  config.theme = config.theme === 'light' ? 'dark' : 'light'
}

function replaceNotifications(): void {
  // 替换整个 notifications 对象，deep: false 也能检测到（引用变了）
  config.notifications = {
    email: false,
    push: true,
    digest: false,
  }
}
</script>

<template>
  <div>
    <p>主题：{{ config.theme }}</p>
    <p>语言：{{ config.language }}</p>
    <p>通知设置：邮件={{ config.notifications.email }}，推送={{ config.notifications.push }}，摘要={{ config.notifications.digest }}</p>
    <p>超时时间：{{ config.timeout }}ms</p>

    <button @click="changeTheme">切换主题</button>
    <button @click="toggleEmail">切换邮件通知</button>
    <button @click="replaceNotifications">替换整个通知设置</button>

    <p>控制台对比：deep: false 只有替换整个 config 或整个 notifications 才触发；deep: true 对嵌套属性的任意修改都会触发</p>
  </div>
</template>
