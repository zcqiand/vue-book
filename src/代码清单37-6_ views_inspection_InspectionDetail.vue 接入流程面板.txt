<template>
  <main class="mx-auto max-w-5xl space-y-6 p-6">
    <header>
      <p class="text-sm font-medium text-slate-500">检测记录</p>
      <h1 class="mt-1 text-3xl font-semibold text-slate-900">记录 {{ recordId }}</h1>
    </header>

    <FlowPanel :record-id="recordId" @refresh-reports="refreshReports" />

    <section class="rounded-2xl border border-slate-200 bg-white p-5 shadow-sm">
      <h2 class="text-xl font-semibold text-slate-900">报告列表</h2>
      <p class="mt-2 text-sm text-slate-500">刷新次数：{{ reportRefreshCount }}</p>
      <p class="mt-3 rounded-xl bg-slate-50 p-4 text-sm leading-6 text-slate-600">
        当流程进入「报告生成」状态时，FlowPanel 会触发 refresh-reports，父组件可以在这里重新请求报告列表。
      </p>
    </section>
  </main>
</template>

<script setup lang="ts">
import { ref } from 'vue';
import FlowPanel from '@/views/inspection/FlowPanel.vue';
import type { Status } from '@/types/workflow';

const props = defineProps<{
  recordId: string;
}>();

const reportRefreshCount = ref<number>(0);
const recordId = props.recordId;

function refreshReports(payload: { recordId: string; status: Status }): void {
  if (payload.recordId !== recordId) {
    return;
  }

  reportRefreshCount.value += 1;
}
</script>
