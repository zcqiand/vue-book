src/
├── assets/              ← 被 import 引用的静态资源（图片、字体、样式）
│   ├── logo.svg
│   ├── base.css
│   └── main.css
├── components/          ← 可复用的通用 UI 组件（按钮、卡片、模态框）
│   ├── BaseButton.vue
│   ├── UserCard.vue
│   └── ModalDialog.vue
├── composables/         ← 组合式函数（use 开头，封装可复用有状态逻辑）
│   ├── useCounter.ts
│   └── useFetch.ts
├── router/              ← 路由表（URL 路径到组件的映射）
│   └── index.ts
├── stores/              ← Pinia 状态仓库（按业务领域拆分）
│   ├── user.ts
│   └── cart.ts
├── views/               ← 页面级组件（通常一条路由对应一个页面）
│   ├── HomeView.vue
│   ├── LoginView.vue
│   └── UserListView.vue
├── App.vue              ← 根组件，组件树的根（留在 src 根目录）
└── main.ts              ← 应用入口（留在 src 根目录）
