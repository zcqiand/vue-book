my-vue-app/
├── index.html          ← 应用入口 HTML
├── package.json        ← 项目依赖和脚本配置
├── vite.config.ts      ← Vite 构建配置
├── tsconfig.json       ← TypeScript 配置（根，引用子配置）
├── tsconfig.app.json   ← TypeScript 配置（应用代码）
├── tsconfig.node.json  ← TypeScript 配置（Node 端配置文件）
└── src/
    ├── main.ts         ← 应用入口：创建应用实例并挂载
    ├── App.vue         ← 根组件：当前看到的页面内容
    ├── assets/         ← 静态资源（图片、图标）
    ├── components/      ← 通用组件
    ├── router/         ← Vue Router 路由配置（勾选 Router 后生成）
    ├── stores/         ← Pinia 状态仓库（勾选 Pinia 后生成）
    └── views/          ← 页面级组件（勾选 Router 后生成）
