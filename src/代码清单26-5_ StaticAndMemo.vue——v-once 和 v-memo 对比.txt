<!-- src/components/StaticAndMemo.vue -->
<template>
  <div class="static-memo-demo">
    <!-- v-once：静态内容，整个生命周期只渲染一次 -->
    <!-- 适合：App 标题、公司名称、帮助文案等永不变化的内容 -->
    <h1 v-once class="app-title">{{ appTitle }}</h1>
    <!-- appTitle 后续变化不会导致 h1 重新渲染 -->

    <p class="company-name" v-once>北京示例科技有限公司</p>

    <!-- 工具栏 -->
    <div class="toolbar">
      <button @click="randomize" class="btn-random">随机数据</button>
      <button @click="toggleSelect(1)" class="btn-toggle">切换第2项选中</button>
    </div>

    <!-- v-memo：大列表项，仅依赖变化时重渲染 -->
    <!-- 适合：长列表中只有部分数据变化的场景 -->
    <!-- v-memo="[item.id, item.selected]" 表示：只有这两个值变化时才重渲染该行 -->
    <div class="list-container">
      <div
        v-for="item in list"
        :key="item.id"
        v-memo="[item.id, item.selected]"
        :class="['list-item', { selected: item.selected }]"
      >
        <span class="item-id">#{{ item.id }}</span>
        <span class="item-name">{{ item.name }}</span>
        <span class="item-score">{{ item.score }}</span>
        <!-- selected 变化时该行重渲染，score 变化时也重渲染 -->
        <span :class="['item-badge', item.selected ? 'badge-active' : 'badge-default']">
          {{ item.selected ? '已选中' : '未选中' }}
        </span>
      </div>
    </div>

    <p class="hint">
      提示：切换「第2项选中」只会重渲染第2行，其他行不受影响。
      相比不加 v-memo，每次 selected 变化都会导致整个列表重渲染。
    </p>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'

const appTitle = ref<string>('性能优化演示')

interface ListItem {
  id: number
  name: string
  score: number
  selected: boolean
}

const list = ref<ListItem[]>([
  { id: 1, name: 'Alice', score: 92, selected: false },
  { id: 2, name: 'Bob', score: 87, selected: false },
  { id: 3, name: 'Charlie', score: 95, selected: false },
  { id: 4, name: 'Diana', score: 78, selected: false },
  { id: 5, name: 'Eve', score: 88, selected: false },
])

function randomize(): void {
  list.value = list.value.map((item) => ({
    ...item,
    score: Math.floor(Math.random() * 100),
  }))
}

function toggleSelect(id: number): void {
  const item = list.value.find((i) => i.id === id)
  if (item) {
    item.selected = !item.selected
  }
}
</script>
