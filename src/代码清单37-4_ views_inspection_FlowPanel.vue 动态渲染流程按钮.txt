<template>
  <section class="rounded-2xl border border-slate-200 bg-white p-5 shadow-sm">
    <div class="flex flex-col gap-3 border-b border-slate-100 pb-4 sm:flex-row sm:items-center sm:justify-between">
      <div>
        <p class="text-sm font-medium text-slate-500">当前流程状态</p>
        <h2 class="mt-1 text-2xl font-semibold text-slate-900">{{ workflow.currentStatus }}</h2>
      </div>

      <div class="text-sm text-slate-500">
        <span v-if="workflow.updatedAt">更新于：{{ formattedUpdatedAt }}</span>
        <span v-else>尚未从服务端加载更新时间</span>
      </div>
    </div>

    <div class="mt-4 rounded-xl bg-slate-50 p-4">
      <p class="text-sm leading-6 text-slate-600">
        可执行动作来自 Pinia 的 <code>availableActions</code>。它同时考虑当前状态和用户角色，避免只在按钮层做表面禁用。
      </p>
    </div>

    <div v-if="visibleError" class="mt-4 rounded-xl border border-red-200 bg-red-50 px-4 py-3 text-sm text-red-700">
      {{ visibleError }}
    </div>

    <div class="mt-5 flex flex-wrap gap-3">
      <button
        v-for="action in workflow.availableActions"
        :key="action.id"
        type="button"
        class="rounded-xl bg-slate-900 px-4 py-2 text-sm font-medium text-white transition hover:bg-slate-700 disabled:cursor-not-allowed disabled:bg-slate-300"
        :disabled="workflow.loading || pendingActionId !== null"
        @click="handleTransition(action)"
      >
        <span v-if="pendingActionId === action.id">处理中...</span>
        <span v-else>{{ action.label }}</span>
      </button>

      <p
        v-if="!workflow.loading && workflow.availableActions.length === 0"
        class="rounded-xl bg-emerald-50 px-4 py-2 text-sm text-emerald-700"
      >
        当前状态没有可执行动作。
      </p>
    </div>

    <div v-if="workflow.loading" class="mt-4 text-sm text-slate-500">
      正在同步流程状态，请稍候。
    </div>
  </section>
</template>

<script setup lang="ts">
import { computed, onMounted, ref, watch } from 'vue';
import { useWorkflowStore } from '@/stores/workflow';
import { Status, type WorkflowAction, type WorkflowActionId } from '@/types/workflow';

const props = defineProps<{
  recordId: string;
}>();

const emit = defineEmits<{
  refreshReports: [{ recordId: string; status: Status }];
}>();

const workflow = useWorkflowStore();
const pendingActionId = ref<WorkflowActionId | null>(null);
const localError = ref<string>('');

const visibleError = computed<string>(() => localError.value || workflow.error);

const formattedUpdatedAt = computed<string>(() => {
  if (!workflow.updatedAt) {
    return '';
  }

  return new Intl.DateTimeFormat('zh-CN', {
    year: 'numeric',
    month: '2-digit',
    day: '2-digit',
    hour: '2-digit',
    minute: '2-digit',
    second: '2-digit',
  }).format(new Date(workflow.updatedAt));
});

function normalizeErrorMessage(cause: unknown): string {
  if (cause instanceof Error) {
    return cause.message;
  }

  if (typeof cause === 'string') {
    return cause;
  }

  return '流程操作失败，请稍后重试';
}

async function handleTransition(action: WorkflowAction): Promise<void> {
  const confirmed = window.confirm(action.confirmMessage);

  if (!confirmed) {
    return;
  }

  pendingActionId.value = action.id;
  localError.value = '';

  try {
    await workflow.transition(props.recordId, action.id);
  } catch (cause) {
    localError.value = normalizeErrorMessage(cause);
  } finally {
    pendingActionId.value = null;
  }
}

onMounted(async () => {
  try {
    await workflow.loadWorkflow(props.recordId);
  } catch (cause) {
    localError.value = normalizeErrorMessage(cause);
  }
});

watch(
  () => workflow.currentStatus,
  (nextStatus, previousStatus) => {
    if (nextStatus === Status.ReportGenerated && previousStatus !== Status.ReportGenerated) {
      // 报告生成后通知父组件刷新列表，是为了让报告区域与流程状态保持同一时刻的数据视图。
      emit('refreshReports', {
        recordId: props.recordId,
        status: nextStatus,
      });
    }
  },
);
</script>
