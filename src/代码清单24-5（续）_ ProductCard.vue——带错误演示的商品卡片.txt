<template>
  <div class="product-card">
    <img :src="`https://picsum.photos/seed/${product.id}/400/200`" alt="商品图片" class="product-img" />
    <div class="product-info">
      <h3>{{ product.name }}</h3>
      <p class="product-desc">{{ product.description }}</p>
      <p class="product-price">¥{{ product.price }}</p>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted } from 'vue'

const props = defineProps<{
  productId: number
}>()

// 商品数据
const product = ref({
  id: props.productId,
  name: '加载中...',
  description: '',
  price: 0,
})

// 模拟异步加载商品数据（随机模拟成功或失败）
onMounted(async () => {
  try {
    // 模拟 500ms 网络延迟
    await new Promise((resolve) => setTimeout(resolve, 500))

    // 模拟随机失败（实际项目中应处理真实接口）
    const shouldFail = Math.random() < 0.3
    if (shouldFail) {
      throw new Error('商品数据加载失败，请稍后重试。')
    }

    product.value = {
      id: props.productId,
      name: `商品 #${props.productId}`,
      description: '这是商品的详细描述信息。',
      price: Math.floor(Math.random() * 1000) + 100,
    }
  } catch (err) {
    // 注意：这个 catch 不能阻止错误向上冒泡
    // 因为 onMounted 的异步错误会被 Suspense 的 #error 插槽捕获
    if (err instanceof Error) {
      throw err // 重新抛出，让 Suspense 处理
    }
    throw new Error(String(err))
  }
})
</script>
