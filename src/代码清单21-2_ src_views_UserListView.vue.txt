<script setup lang="ts">
import { useFetch } from '../composables/useFetch'

interface User {
  id: number
  name: string
  email: string
}

const { data: users, error, loading } = useFetch<User[]>('/api/users')
</script>

<template>
  <div class="user-list">
    <h2>用户列表</h2>

    <!-- 加载态 -->
    <div v-if="loading" class="loading">
      <span>加载中...</span>
    </div>

    <!-- 错误态 -->
    <div v-else-if="error" class="error">
      <p>加载失败：{{ error.message }}</p>
      <button @click="$router.go(0)">重新加载</button>
    </div>

    <!-- 成功态 -->
    <ul v-else-if="users" class="user-items">
      <li v-for="user in users" :key="user.id">
        {{ user.name }} - {{ user.email }}
      </li>
    </ul>
  </div>
</template>

<style scoped>
.user-list { max-width: 600px; margin: 0 auto; padding: 24px; }
.loading { padding: 20px; text-align: center; color: #6b7280; }
.error { padding: 20px; color: #dc2626; }
.user-items { list-style: none; padding: 0; }
.user-items li { padding: 8px 0; border-bottom: 1px solid #e5e7eb; }
</style>
