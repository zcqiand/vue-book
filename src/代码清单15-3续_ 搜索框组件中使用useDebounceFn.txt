<!-- src/components/SearchBox.vue -->
<script setup lang="ts">
import { ref } from 'vue'
import { useDebounceFn } from '../composables/useDebounceFn'

const keyword = ref<string>('')
const results = ref<string[]>([])

// searchAPI 是真实接口，这里用 setTimeout 模拟异步返回
function searchAPI(query: string): void {
  console.log('实际发起搜索：', query)
  results.value = [`${query} 的结果1`, `${query} 的结果2`]
}

// 传入 500ms 防抖延迟，连续输入只会在停笔后触发一次搜索
const debouncedSearch = useDebounceFn(searchAPI, 500)

function onInput(value: string): void {
  keyword.value = value
  debouncedSearch(value)
}
</script>

<template>
  <div>
    <input :value="keyword" @input="(e) => onInput((e.target as HTMLInputElement).value)" placeholder="输入关键词搜索" />
    <ul>
      <li v-for="item in results" :key="item">{{ item }}</li>
    </ul>
  </div>
</template>
