<template>
  <div class="error-demo">
    <h2>错误处理对比</h2>

    <!-- 场景一：事件处理函数中的 await 错误——用 try-catch 接 -->
    <div class="demo-section">
      <h3>场景一：API 请求错误（try-catch）</h3>
      <p>点击按钮模拟网络请求失败，try-catch 捕获异步错误。</p>
      <button class="btn-action" @click="fetchData">请求数据</button>
      <p v-if="apiError" class="error-msg">{{ apiError }}</p>
    </div>

    <!-- 场景二：子组件渲染错误——用 onErrorCaptured 接 -->
    <div class="demo-section">
      <h3>场景二：子组件渲染错误（onErrorCaptured）</h3>
      <p>切换开关触发子组件渲染错误，由 ErrorBoundary 的 onErrorCaptured 捕获。</p>
      <button class="btn-toggle" @click="toggleChild">切换子组件</button>

      <ErrorBoundary>
        <FlakyChild v-if="showChild" :shouldFail="true" />
      </ErrorBoundary>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import ErrorBoundary from '../components/ErrorBoundary.vue'
import FlakyChild from '../components/FlakyChild.vue'

// ========== 场景一：事件处理函数中的 await 错误 ==========
// 使用 try-catch 接住 fetch 请求的网络错误和业务错误
const apiError = ref<string | null>(null)

async function fetchData(): Promise<void> {
  apiError.value = null

  try {
    // 模拟一个会失败的 fetch 请求
    const response = await fetch('/api/nonexistent-endpoint')

    if (!response.ok) {
      // HTTP 错误（404、500 等），主动抛出
      throw new Error(`请求失败: ${response.status} ${response.statusText}`)
    }

    const data = await response.json()
    console.log('数据获取成功:', data)
  } catch (err) {
    // catch 块捕获 try 中的所有同步和异步错误
    if (err instanceof Error) {
      apiError.value = err.message
    } else {
      apiError.value = String(err)
    }
  }
}

// ========== 场景二：子组件渲染错误——onErrorCaptured 接 ==========
// onErrorCaptured 专门处理来自子组件树中的渲染错误和侦听器错误
const showChild = ref(false)

function toggleChild(): void {
  showChild.value = !showChild.value
}
</script>
