<template>
  <div class="table-view">
    <h2 class="table-title">数据表（共 {{ rowCount }} 行）</h2>

    <!-- 工具栏 -->
    <div class="toolbar">
      <button @click="loadData" class="btn-load" :disabled="loading">
        {{ loading ? '加载中...' : '刷新数据' }}
      </button>
      <span class="hint">使用 shallowRef，只在整体替换时触发渲染</span>
    </div>

    <!-- 表格内容 -->
    <table v-if="tableData.value.length > 0" class="data-table">
      <thead>
        <tr>
          <th v-for="col in columns" :key="col">{{ col }}</th>
        </tr>
      </thead>
      <tbody>
        <!-- 注意：这里遍历 tableData.value，而非 tableData -->
        <tr v-for="(row, idx) in tableData.value" :key="idx">
          <td v-for="col in columns" :key="col">{{ row[col] }}</td>
        </tr>
      </tbody>
    </table>

    <div v-else class="empty-state">点击「刷新数据」加载表格</div>
  </div>
</template>

<script setup lang="ts">
import { shallowRef } from 'vue'

// 错误写法（注释掉）：
// const tableData = ref<Row[]>([])  // ref 会深层代理，每行每个单元格都追踪，太慢
// 加载万行数据时，每次 tableData.value.push(row) 都会触发 Proxy 链路追踪

// 正确写法：shallowRef 只在 .value 整体替换时触发一次渲染
const tableData = shallowRef<Row[]>([])
const loading = ref<boolean>(false)

const columns = ['id', 'name', 'email', 'department', 'salary'] as const

interface Row {
  id: number
  name: string
  email: string
  department: string
  salary: number
}

async function loadData(): Promise<void> {
  loading.value = true
  try {
    // 模拟接口返回万行数据
    const res = await fetchTableData()
    // 关键：整体替换 .value，而不是逐行 push
    // shallowRef 只比较 .value 的引用是否变化，不追踪深层属性
    tableData.value = res.data
  } finally {
    loading.value = false
  }
}

// 模拟 fetch
async function fetchTableData(): Promise<{ data: Row[] }> {
  await new Promise((resolve) => setTimeout(resolve, 800))
  const rows: Row[] = Array.from({ length: 10000 }, (_, i) => ({
    id: i + 1,
    name: `员工 ${i + 1}`,
    email: `user${i + 1}@example.com`,
    department: ['技术部', '产品部', '运营部'][i % 3],
    salary: 8000 + Math.floor(Math.random() * 12000),
  }))
  return { data: rows }
}

// 计算行数（响应式访问 .value）
const rowCount = computed(() => tableData.value.length)
</script>
