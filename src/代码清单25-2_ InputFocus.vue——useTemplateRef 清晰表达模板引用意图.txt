<!-- src/components/InputFocus.vue -->
<template>
  <div class="input-focus">
    <!-- 旧写法：ref 属性值需要和 script 中变量名一致 -->
    <!-- useTemplateRef 新写法：ref 属性值作为字符串参数传入 -->
    <input ref="input-el" type="text" placeholder="点击按钮自动聚焦" class="text-input" />

    <button @click="focus" class="btn-focus">聚焦输入框</button>
    <button @click="blur" class="btn-blur">失焦输入框</button>
  </div>
</template>

<script setup lang="ts">
import { useTemplateRef } from 'vue'

// useTemplateRef<HTMLInputElement>('input-el')
// 参数 'input-el' 必须和模板中 ref="input-el" 的值完全一致
// 返回 Ref<HTMLInputElement | null>，指向模板中的 input DOM 节点
const inputRef = useTemplateRef<HTMLInputElement>('input-el')

function focus(): void {
  // inputRef.value 是 Ref，直接访问 .value 获取 DOM 节点
  inputRef.value?.focus()
}

function blur(): void {
  inputRef.value?.blur()
}
</script>
