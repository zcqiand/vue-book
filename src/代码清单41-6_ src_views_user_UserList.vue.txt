<script setup lang="ts">
import { computed, onMounted, ref } from 'vue'
import OrgTreeNode from '@/components/OrgTreeNode.vue'
import { useResource } from '@/composables/useResource'
import { useTable } from '@/composables/useTable'
import { useUserStore, type OrgNode, type User, type UserInput } from '@/stores/user'

const userStore = useUserStore()
const selectedOrgId = ref('')
const actionError = ref('')
const roleDialogVisible = ref(false)
const roleEditingUser = ref<User | null>(null)
const roleEditingIds = ref<string[]>([])

const table = useTable<User, { keyword: string; orgId: string; status: '' | 'enabled' | 'disabled' }>(
  (query) => userStore.listUsers(query),
  { keyword: '', orgId: '', status: '' },
  8
)

const resource = useResource<UserInput, string>({
  createEmpty: () => ({
    orgId: selectedOrgId.value || 'org-frontend',
    username: '',
    displayName: '',
    email: '',
    roleIds: ['role-viewer'],
    status: 'enabled'
  }),
  create: (payload) => userStore.createUser(payload),
  update: (id, payload) => userStore.updateUser(id, payload),
  afterSaved: () => table.reload()
})

const selectedOrgName = computed(() => selectedOrgId.value || '全部组织')

function selectOrg(node: OrgNode): void {
  selectedOrgId.value = node.id
  table.state.query.orgId = node.id
  void table.search()
}

async function clearOrg(): Promise<void> {
  selectedOrgId.value = ''
  table.state.query.orgId = ''
  await table.search()
}

function openEdit(user: User): void {
  resource.openEdit(user.id, {
    orgId: user.orgId,
    username: user.username,
    displayName: user.displayName,
    email: user.email,
    roleIds: [...user.roleIds],
    status: user.status
  })
}

function openRoleDialog(user: User): void {
  roleEditingUser.value = user
  roleEditingIds.value = [...user.roleIds]
  roleDialogVisible.value = true
}

async function saveRoles(): Promise<void> {
  if (!roleEditingUser.value) {
    return
  }
  actionError.value = ''
  try {
    await userStore.assignRoles(roleEditingUser.value.id, roleEditingIds.value)
    roleDialogVisible.value = false
    await table.reload()
  } catch (caught) {
    actionError.value = caught instanceof Error ? caught.message : '角色分配失败'
  }
}

async function toggleStatus(user: User): Promise<void> {
  actionError.value = ''
  try {
    await userStore.toggleStatus(user.id)
    await table.reload()
  } catch (caught) {
    actionError.value = caught instanceof Error ? caught.message : '状态切换失败'
  }
}

async function removeUser(user: User): Promise<void> {
  actionError.value = ''
  try {
    await userStore.deleteUser(user.id)
    await table.reload()
  } catch (caught) {
    actionError.value = caught instanceof Error ? caught.message : '删除用户失败'
  }
}

function roleNames(user: User): string {
  return user.roleIds.map((id) => userStore.roleNameMap.get(id) ?? id).join('、')
}

onMounted(() => {
  void table.reload()
})
</script>

<template>
  <section class="grid gap-4 lg:grid-cols-[260px_1fr]">
    <aside class="rounded-2xl border border-slate-200 bg-white p-4 shadow-sm">
      <div class="mb-3 flex items-center justify-between">
        <h2 class="font-semibold text-slate-900">组织架构</h2>
        <button type="button" class="text-sm text-blue-600 hover:text-blue-700" @click="clearOrg">全部</button>
      </div>
      <ul class="space-y-1">
        <OrgTreeNode
          v-for="node in userStore.orgTree"
          :key="node.id"
          :node="node"
          :selected-id="selectedOrgId"
          @select="selectOrg"
        />
      </ul>
    </aside>

    <main class="rounded-2xl border border-slate-200 bg-white p-4 shadow-sm">
      <div class="mb-4 flex flex-col gap-3 xl:flex-row xl:items-end xl:justify-between">
        <div>
          <h1 class="text-xl font-semibold text-slate-900">用户管理</h1>
          <p class="text-sm text-slate-500">当前筛选：{{ selectedOrgName }}</p>
        </div>
        <form class="flex flex-wrap gap-2" @submit.prevent="table.search">
          <input v-model="table.state.query.keyword" class="rounded-lg border border-slate-300 px-3 py-2 text-sm" placeholder="搜索用户名/姓名/邮箱" />
          <select v-model="table.state.query.status" class="rounded-lg border border-slate-300 px-3 py-2 text-sm">
            <option value="">全部状态</option>
            <option value="enabled">启用</option>
            <option value="disabled">禁用</option>
          </select>
          <button type="submit" class="rounded-lg bg-slate-900 px-4 py-2 text-sm text-white">搜索</button>
          <button type="button" class="rounded-lg bg-blue-600 px-4 py-2 text-sm text-white" @click="resource.openCreate">新增用户</button>
        </form>
      </div>

      <p v-if="table.error || actionError" class="mb-3 rounded-lg bg-red-50 px-3 py-2 text-sm text-red-700">
        {{ table.error || actionError }}
      </p>

      <div class="overflow-x-auto">
        <table class="min-w-full divide-y divide-slate-200 text-sm">
          <thead class="bg-slate-50 text-left text-slate-600">
            <tr>
              <th class="px-3 py-2">用户名</th>
              <th class="px-3 py-2">姓名</th>
              <th class="px-3 py-2">邮箱</th>
              <th class="px-3 py-2">角色</th>
              <th class="px-3 py-2">状态</th>
              <th class="px-3 py-2">操作</th>
            </tr>
          </thead>
          <tbody class="divide-y divide-slate-100">
            <tr v-for="user in table.rows" :key="user.id" class="hover:bg-slate-50">
              <td class="px-3 py-2 font-medium text-slate-900">{{ user.username }}</td>
              <td class="px-3 py-2">{{ user.displayName }}</td>
              <td class="px-3 py-2">{{ user.email }}</td>
              <td class="px-3 py-2">{{ roleNames(user) }}</td>
              <td class="px-3 py-2">
                <span class="rounded-full px-2 py-1 text-xs" :class="user.status === 'enabled' ? 'bg-green-50 text-green-700' : 'bg-slate-100 text-slate-600'">
                  {{ user.status === 'enabled' ? '启用' : '禁用' }}
                </span>
              </td>
              <td class="space-x-2 px-3 py-2">
                <button type="button" class="text-blue-600" @click="openEdit(user)">编辑</button>
                <button type="button" class="text-indigo-600" @click="openRoleDialog(user)">角色</button>
                <button type="button" class="text-amber-600" @click="toggleStatus(user)">{{ user.status === 'enabled' ? '禁用' : '启用' }}</button>
                <button type="button" class="text-red-600" @click="removeUser(user)">删除</button>
              </td>
            </tr>
          </tbody>
        </table>
      </div>

      <div class="mt-4 flex items-center justify-between text-sm text-slate-600">
        <span>共 {{ table.state.total }} 条，第 {{ table.state.page }} / {{ table.pageCount }} 页</span>
        <div class="space-x-2">
          <button type="button" class="rounded border px-3 py-1 disabled:opacity-40" :disabled="table.state.page <= 1" @click="table.previousPage">上一页</button>
          <button type="button" class="rounded border px-3 py-1 disabled:opacity-40" :disabled="table.state.page >= table.pageCount" @click="table.nextPage">下一页</button>
        </div>
      </div>
    </main>

    <div v-if="resource.visible" class="fixed inset-0 z-40 grid place-items-center bg-slate-900/40 p-4">
      <form class="w-full max-w-lg rounded-2xl bg-white p-5 shadow-xl" @submit.prevent="resource.save">
        <h2 class="mb-4 text-lg font-semibold">{{ resource.editingId ? '编辑用户' : '新增用户' }}</h2>
        <p v-if="resource.error" class="mb-3 rounded bg-red-50 px-3 py-2 text-sm text-red-700">{{ resource.error }}</p>
        <div class="grid gap-3">
          <input v-model="resource.form.username" required class="rounded-lg border px-3 py-2" placeholder="用户名" />
          <input v-model="resource.form.displayName" required class="rounded-lg border px-3 py-2" placeholder="姓名" />
          <input v-model="resource.form.email" required type="email" class="rounded-lg border px-3 py-2" placeholder="邮箱" />
          <select v-model="resource.form.orgId" class="rounded-lg border px-3 py-2">
            <option value="org-frontend">前端组</option>
            <option value="org-backend">后端组</option>
            <option value="org-cs">客户成功组</option>
          </select>
          <select v-model="resource.form.status" class="rounded-lg border px-3 py-2">
            <option value="enabled">启用</option>
            <option value="disabled">禁用</option>
          </select>
          <label v-for="role in userStore.roles" :key="role.id" class="flex items-center gap-2 text-sm">
            <input v-model="resource.form.roleIds" type="checkbox" :value="role.id" />
            {{ role.name }}
          </label>
        </div>
        <div class="mt-5 flex justify-end gap-2">
          <button type="button" class="rounded-lg border px-4 py-2" @click="resource.close">取消</button>
          <button type="submit" class="rounded-lg bg-blue-600 px-4 py-2 text-white" :disabled="resource.saving">保存</button>
        </div>
      </form>
    </div>

    <div v-if="roleDialogVisible" class="fixed inset-0 z-40 grid place-items-center bg-slate-900/40 p-4">
      <form class="w-full max-w-md rounded-2xl bg-white p-5 shadow-xl" @submit.prevent="saveRoles">
        <h2 class="mb-4 text-lg font-semibold">分配角色：{{ roleEditingUser?.displayName }}</h2>
        <label v-for="role in userStore.roles" :key="role.id" class="mb-2 flex items-center gap-2 text-sm">
          <input v-model="roleEditingIds" type="checkbox" :value="role.id" />
          {{ role.name }}
        </label>
        <div class="mt-5 flex justify-end gap-2">
          <button type="button" class="rounded-lg border px-4 py-2" @click="roleDialogVisible = false">取消</button>
          <button type="submit" class="rounded-lg bg-indigo-600 px-4 py-2 text-white">保存角色</button>
        </div>
      </form>
    </div>
  </section>
</template>
