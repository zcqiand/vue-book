my-vue-app/
├── public/                  ← 静态资源，构建时原样复制到 dist/
│   └── favicon.ico          ← 站点图标，不参与编译
├── src/                     ← 应用源码，业务代码全部住在这里
│   ├── assets/              ← 被 import 引用的资源（图片、字体、样式）
│   │   ├── base.css         ← 基础样式（CSS 变量、重置样式）
│   │   ├── main.css         ← 主样式入口，被 main.ts 引入
│   │   └── logo.svg         ← 被 App.vue import 的 Logo
│   ├── components/          ← 通用 UI 组件（可跨页面复用）
│   │   └── TheWelcome.vue   ← 模板生成的示例组件
│   ├── router/              ← Vue Router 路由表（勾选 Router 后生成）
│   │   └── index.ts         ← 路由配置：URL 路径到组件的映射
│   ├── stores/              ← Pinia 状态仓库（勾选 Pinia 后生成）
│   │   └── counter.ts       ← 示例 store
│   ├── views/               ← 页面级组件（勾选 Router 后生成）
│   │   ├── AboutView.vue    ← 示例页面：关于页
│   │   └── HomeView.vue     ← 示例页面：首页
│   ├── App.vue              ← 根组件，组件树的根
│   └── main.ts              ← 应用入口：创建实例、注册插件、挂载
├── .gitignore               ← Git 忽略规则（忽略 node_modules、dist）
├── eslint.config.js         ← ESLint 9.x 代码规范配置（flat config）
├── index.html               ← 应用入口 HTML 页面（Vite 的真正起点）
├── package.json             ← 项目元信息、依赖、脚本命令
├── README.md                ← 项目说明文档
├── tsconfig.app.json        ← TypeScript 配置（应用源码 src/）
├── tsconfig.json            ← TypeScript 配置（根，references 引用子配置）
├── tsconfig.node.json       ← TypeScript 配置（Node 端配置文件）
└── vite.config.ts           ← Vite 构建配置（插件、开发服务器）
