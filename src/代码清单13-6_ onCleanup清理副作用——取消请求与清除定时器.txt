<script setup lang="ts">
import { ref, watch, onCleanup } from 'vue'

const userId = ref<string>('')
const userInfo = ref<{ name: string; bio: string } | null>(null)
const loading = ref<boolean>(false)

let requestTimer: ReturnType<typeof setTimeout> | null = null

watch(
  userId,
  async (newId: string, oldId: string | undefined) => {
    if (!newId.trim()) {
      userInfo.value = null
      loading.value = false
      return
    }

    // 每次 watch 重新执行前，清理上一次的定时器和请求
    // 这确保了：快速切换 userId 时，上一个请求/定时器不会在完成后覆盖最新结果
    if (requestTimer !== null) {
      clearTimeout(requestTimer)
      requestTimer = null
      console.log(`[cleanup] 取消上一个定时器（从 ${oldId ?? '无'} 切换到 ${newId}）`)
    }

    loading.value = true
    console.log(`[search] 开始查询用户: ${newId}`)

    // 模拟带取消的异步请求：用 setTimeout + 清理函数模拟 AbortController
    const currentId = newId // 闭包捕获当前请求对应的 id
    requestTimer = setTimeout(async () => {
      // 请求发出后检查是否已切换到其他 userId
      // 如果已切换，则丢弃结果（通过 requestTimer 是否为 null 判断是否被取消）
      if (requestTimer === null) {
        console.log(`[abandoned] 用户 ${currentId} 的请求已被取消`)
        return
      }

      // 模拟请求完成
      const mockData: { name: string; bio: string } = {
        name: `用户${currentId}`,
        bio: `这是用户 ${currentId} 的个人简介`,
      }

      // 再次检查：请求完成时 userId 是否已切换（竞态检查）
      if (userId.value !== currentId) {
        console.log(`[abandoned] 用户 ${currentId} 请求完成但 userId 已切换为 ${userId.value}`)
        return
      }

      userInfo.value = mockData
      loading.value = false
      console.log(`[success] 用户 ${currentId} 信息加载完成`)
      requestTimer = null // 成功后清除标记
    }, 800)
  },
  { immediate: true }
)

// 组件卸载时清理所有定时器
onCleanup(() => {
  if (requestTimer !== null) {
    clearTimeout(requestTimer)
    requestTimer = null
    console.log('[cleanup] 组件卸载，清除定时器')
  }
})

function loadUser(id: string): void {
  userId.value = id
}

function clearUser(): void {
  userId.value = ''
}
</script>

<template>
  <div>
    <input v-model="userId" type="text" placeholder="输入用户ID" />
    <button @click="loadUser('u001')">加载 u001</button>
    <button @click="loadUser('u002')">加载 u002</button>
    <button @click="loadUser('u003')">加载 u003</button>
    <button @click="clearUser">清空</button>

    <p>加载状态：{{ loading ? '加载中...' : '空闲' }}</p>
    <div v-if="userInfo">
      <p>姓名：{{ userInfo.name }}</p>
      <p>简介：{{ userInfo.bio }}</p>
    </div>

    <p>快速切换用户ID，观察控制台：被丢弃的请求会打印 abandoned，已完成的请求结果不会覆盖当前结果</p>
  </div>
</template>
