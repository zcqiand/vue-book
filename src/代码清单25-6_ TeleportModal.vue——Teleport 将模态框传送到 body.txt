<template>

    </button>

    <!-- Teleport: to="body" 将内部内容传送到 body 末尾 -->
    <!-- 内容虽然在 body 下，但事件处理、响应式状态仍在当前组件内 -->
    <Teleport to="body">
      <div v-if="show" class="modal-overlay">
        <div class="modal">
          <h3>Teleport 模态框</h3>
          <p>这个模态框已经被 Teleport 传送到了 body 下。</p>
          <p>它不再受父级元素的 overflow / z-index 影响。</p>
          <button @click="show = false" class="btn-close">关闭</button>
        </div>
      </div>
    </Teleport>
  </div>
</template>
