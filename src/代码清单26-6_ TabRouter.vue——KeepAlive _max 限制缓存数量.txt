<template>
  <div class="tab-router">
    <h2 class="section-title">KeepAlive 缓存（:max=3）</h2>

    <!-- Tab 切换按钮 -->
    <div class="tab-bar">
      <button
        v-for="tab in tabs"
        :key="tab.name"
        :class="['tab-btn', { active: currentTab === tab.name }]"
        @click="switchTab(tab.name)"
      >
        {{ tab.label }}
        <span class="tab-badge">{{ tab.cacheInfo }}</span>
      </button>
    </div>

    <!-- KeepAlive :max=3：最多缓存 3 个组件实例 -->
    <!-- 第 4 个组件被缓存时，最久未使用的会被销毁 -->
    <!-- 适合：Tab 数量不确定、内存敏感的场景 -->
    <KeepAlive :max="3">
      <component :is="currentComponent" :key="currentTab" />
    </KeepAlive>

    <p class="hint">
      提示：打开 4 个以上 Tab，观察前几个 Tab 的缓存状态变化。
      超出 :max 数量后，最早缓存的实例会被 LRU 驱逐。
    </p>
  </div>
</template>

<script setup lang="ts">
import { ref, computed } from 'vue'
import DashboardTab from './DashboardTab.vue'
import AnalyticsTab from './AnalyticsTab.vue'
import SettingsTab from './SettingsTab.vue'
import UsersTab from './UsersTab.vue'
import ReportsTab from './ReportsTab.vue'

interface Tab {
  name: string
  label: string
  component: object
  cacheInfo: string
}

const tabs: Tab[] = [
  { name: 'dashboard', label: '仪表盘', component: DashboardTab, cacheInfo: '' },
  { name: 'analytics', label: '数据分析', component: AnalyticsTab, cacheInfo: '' },
  { name: 'settings', label: '设置', component: SettingsTab, cacheInfo: '' },
  { name: 'users', label: '用户管理', component: UsersTab, cacheInfo: '' },
  { name: 'reports', label: '报表', component: ReportsTab, cacheInfo: '' },
]

const currentTab = ref<string>('dashboard')

const currentComponent = computed(() => {
  const tab = tabs.find((t) => t.name === currentTab.value)
  return tab?.component
})

function switchTab(name: string): void {
  currentTab.value = name
}
</script>
