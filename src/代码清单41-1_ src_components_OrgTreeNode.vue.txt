<script setup lang="ts">
import { computed, ref } from 'vue'
import type { OrgNode } from '@/stores/user'

defineOptions({ name: 'OrgTreeNode' })

const props = defineProps<{ node: OrgNode; selectedId: string }>()
const emit = defineEmits<{ select: [node: OrgNode] }>()

const expanded = ref(true)
const hasChildren = computed(() => props.node.children.length > 0)
const isSelected = computed(() => props.selectedId === props.node.id)

function toggle(): void {
  if (hasChildren.value) expanded.value = !expanded.value
}
</script>

<template>
  <li>
    <div
      class="flex items-center gap-2 rounded px-2 py-1"
      :class="isSelected ? 'bg-blue-50 text-blue-700' : 'hover:bg-slate-100'"
    >
      <button type="button" :disabled="!hasChildren" @click="toggle">
        {{ hasChildren ? (expanded ? '−' : '+') : '·' }}
      </button>
      <button type="button" class="flex-1 text-left" @click="emit('select', node)">
        {{ node.name }}
      </button>
    </div>
    <ul v-if="expanded && hasChildren" class="ml-5 border-l pl-2">
      <OrgTreeNode
        v-for="child in node.children"
        :key="child.id"
        :node="child"
        :selected-id="selectedId"
        @select="emit('select', $event)"
      />
    </ul>
  </li>
</template>
