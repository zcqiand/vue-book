<!-- src/components/CurrentUser.vue -->
<script setup lang="ts">
import { ref, onMounted } from 'vue'

interface User {
  id: number
  name: string
}

// 泛型写 User | null：明确告诉 TS「现在是 null，以后会变成 User」
// 若只写 ref(null)，推断结果是 Ref<null>，后续 user.value = {...} 会类型报错
const user = ref<User | null>(null)
const loading = ref<boolean>(true)

function loadUser(): void {
  // 模拟异步请求：延迟赋值，体现「初始 null、运行时才有值」的常见场景
  window.setTimeout(() => {
    user.value = { id: 1, name: '王五' }
    loading.value = false
  }, 500)
}

onMounted(loadUser)
</script>

<template>
  <div>
    <p v-if="loading">加载中……</p>
    <!-- 模板自动解包：user 为 null 时链式访问会安全跳过（v-if 已拦截） -->
    <p v-else>当前用户：{{ user?.name }}（ID: {{ user?.id }}）</p>
    <button @click="loadUser" :disabled="loading">重新加载</button>
  </div>
</template>
