<script setup lang="ts">
import { ref } from 'vue'

interface User {
  id: number
  name: string
  bio: string
}

const user = ref<User | null>(null)

async function loadUser(): Promise<void> {
  // 模拟网络延迟
  await new Promise((resolve) => setTimeout(resolve, 1500))
  user.value = {
    id: 1,
    name: '张三',
    bio: '全栈工程师，热爱 Vue 和 TypeScript',
  }
}

await loadUser()
</script>

<template>
  <div class="user-profile">
    <h2>{{ user?.name }}</h2>
    <p>{{ user?.bio }}</p>
  </div>
</template>
