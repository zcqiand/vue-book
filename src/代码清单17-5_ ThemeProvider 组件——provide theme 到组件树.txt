<template>
  <div class="theme-provider" :class="themeClass">
    <!-- <slot> 让消费组件正常渲染 -->
    <slot />
  </div>
</template>

<script setup lang="ts">
import { provide, computed, ref } from 'vue'

type ThemeName = 'light' | 'dark' | 'ocean'

// 主题配置：定义每个主题的视觉变量
const themeConfig: Record<ThemeName, Record<string, string>> = {
  light: {
    '--bg-color': '#ffffff',
    '--text-color': '#111827',
    '--accent-color': '#3b82f6',
    '--border-color': '#e5e7eb',
  },
  dark: {
    '--bg-color': '#1f2937',
    '--text-color': '#f9fafb',
    '--accent-color': '#60a5fa',
    '--border-color': '#374151',
  },
  ocean: {
    '--bg-color': '#0f172a',
    '--text-color': '#e0f2fe',
    '--accent-color': '#06b6d4',
    '--border-color': '#155e75',
  },
}

// 当前主题状态
const currentTheme = ref<ThemeName>('light')

// 提供主题名和主题配置（子孙组件可以按需取用）
provide('themeName', currentTheme)
provide('themeConfig', computed(() => themeConfig[currentTheme.value]))

// 切换主题的方法也一并提供下去
function switchTheme(name: ThemeName): void {
  currentTheme.value = name
}
provide('switchTheme', switchTheme)

// 计算顶层容器类名
const themeClass = computed(() => `theme-${currentTheme.value}`)
</script>

<style scoped>
.theme-provider {
  padding: 16px;
  border-radius: 8px;
  transition: background-color 0.3s, color 0.3s;
}

/* 应用 provide 下来的 CSS 变量 */
.theme-provider {
  background-color: var(--bg-color);
  color: var(--text-color);
}
</style>
