<!-- src/components/EditableCounter.vue -->
<script setup lang="ts">
import { ref } from 'vue'

const count = ref(0) // 内部状态，子组件自管

function increment() {
  count.value++
}

// 暴露给父组件的方法：父组件能通过模板引用调用 reset()
function reset() {
  count.value = 0
}

// defineExpose 是编译器宏：声明本组件对外暴露的公共接口
defineExpose({ reset })
</script>

<template>
  <div>
    <span>{{ count }}</span>
    <button type="button" @click="increment">+1</button>
  </div>
</template>
