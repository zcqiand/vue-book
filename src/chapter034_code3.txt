src/
├── main.ts                     # 应用入口：createApp + 注册 Pinia/Router + 挂载 #app
├── App.vue                     # 根组件：仅放 <router-view>，不放任何业务逻辑
├── assets/                     # 静态资源：图片、字体、全局样式入口
│   └── styles/
│       └── main.css            # Tailwind CSS 4 入口（@import "tailwindcss"）
├── components/                 # 跨页面复用的通用组件
│   ├── BaseButton.vue          # 统一按钮：变体/尺寸/加载态，避免每个页面各写一套
│   ├── BaseTable.vue           # 统一表格：分页、排序、空态封装
│   └── form/                   # 表单原子组件：按子目录归类，避免 components 根目录爆炸
│       ├── FormInput.vue
│       └── FormSelect.vue
├── layouts/                    # 页面级布局：定义「外壳」，内容由嵌套路由填充
│   └── MainLayout.vue          # 主布局：顶部导航 + 侧边栏 + <router-view>
├── views/                      # 页面组件：每个文件对应一个路由，按业务模块分子目录
│   ├── auth/
│   │   └── LoginView.vue       # 登录页：独立顶级路由，不进入 MainLayout
│   ├── inspection/             # 检测项目管理模块
│   │   ├── InspectionListView.vue   # 列表页
│   │   └── InspectionDetailView.vue # 详情页：对应动态路由 /inspection/:id
│   ├── sample/                 # 样品管理模块
│   │   └── SampleListView.vue
│   └── report/                 # 报告管理模块
│       └── ReportListView.vue
├── stores/                     # Pinia 状态：按业务模块拆分，每个文件一个 setup store
│   ├── auth.ts                 # 登录态、当前用户、token
│   ├── inspection.ts           # 检测项目列表与详情
│   ├── sample.ts               # 样品数据
│   └── report.ts               # 报告数据
├── router/
│   └── index.ts                # 路由表：createRouter + createWebHistory + 嵌套路由
├── api/                        # 接口层：按业务模块拆分，统一 http 封装
│   ├── http.ts                 # axios 实例：拦截器统一注入 token 与处理错误码
│   ├── auth.ts                 # 登录、登出、获取当前用户的接口封装
│   ├── inspection.ts           # 检测项目接口
│   ├── sample.ts               # 样品接口
│   └── report.ts               # 报告接口
├── composables/                # 可复用的组合式逻辑：跨组件跨页面共享
│   ├── usePagination.ts        # 分页逻辑：列表页通用
│   └── usePermission.ts        # 权限判断：基于 store 的角色计算
├── types/                      # 全局 TypeScript 类型定义
│   ├── api.ts                  # 接口响应/请求的统一类型
│   └── models.ts               # 业务实体类型：User、Inspection、Sample、Report
└── utils/                      # 纯函数工具：无状态、无副作用，便于单测
    ├── format.ts               # 格式化：日期、金额、状态枚举
    └── storage.ts              # localStorage 封装：带前缀与过期时间
