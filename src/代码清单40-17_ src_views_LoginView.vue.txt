<script setup lang="ts">
import { computed } from 'vue'
import { useRouter, useRoute } from 'vue-router'
import { useSso } from '@/composables/useSso'
import { useOAuth } from '@/composables/useOAuth'

const router = useRouter()
const route = useRoute()
const sso = useSso()
const githubOAuth = useOAuth('github')

const returnTo = computed(() => {
  const redirect = route.query.redirect
  return typeof redirect === 'string' && redirect.length > 0 ? redirect : '/'
})

const ssoUrl = computed(() => sso.buildSsoLoginUrl(returnTo.value))
const githubUrl = computed(() => githubOAuth.buildAuthorizationUrl(returnTo.value))

async function simulateSsoLogin(): Promise<void> {
  await router.push(sso.createDemoSsoCallbackUrl(returnTo.value))
}

async function simulateGithubLogin(): Promise<void> {
  await router.push(githubOAuth.createDemoOAuthCallbackUrl(returnTo.value))
}
</script>

<template>
  <section class="rounded-xl border border-slate-200 bg-white p-6 shadow-sm">
    <h2 class="text-lg font-semibold">登录到 SaaS 管理台</h2>
    <p class="mt-2 text-sm leading-6 text-slate-600">
      SSO 与 OAuth 都不会让前端保存第三方密钥。前端只接收回调参数，再调用本系统后端换取业务 token。
    </p>

    <div class="mt-6 grid gap-4 md:grid-cols-2">
      <article class="rounded-lg border border-slate-200 p-4">
        <h3 class="font-medium">企业 SSO</h3>
        <p class="mt-2 break-all text-xs text-slate-500">{{ ssoUrl }}</p>
        <button
          class="mt-4 rounded-lg bg-blue-600 px-4 py-2 text-sm font-medium text-white"
          type="button"
          @click="simulateSsoLogin"
        >
          模拟 SSO 回调登录
        </button>
      </article>

      <article class="rounded-lg border border-slate-200 p-4">
        <h3 class="font-medium">GitHub OAuth2.0</h3>
        <p class="mt-2 break-all text-xs text-slate-500">{{ githubUrl }}</p>
        <button
          class="mt-4 rounded-lg bg-slate-900 px-4 py-2 text-sm font-medium text-white"
          type="button"
          @click="simulateGithubLogin"
        >
          模拟 GitHub 授权码回调
        </button>
      </article>
    </div>
  </section>
</template>
