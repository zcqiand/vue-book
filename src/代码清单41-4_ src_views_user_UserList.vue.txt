<script setup lang="ts">
import { onMounted, ref } from 'vue'
import OrgTreeNode from '@/components/OrgTreeNode.vue'
import { useResource, useTable } from '@/composables/tableAndResource'
import { useUserStore, type OrgNode, type User, type UserInput } from '@/stores/user'

const store = useUserStore()
const selectedOrgId = ref('')
const roleUser = ref<User | null>(null)
const roleIds = ref<string[]>([])
const table = useTable<User, { keyword: string; orgId: string; status: '' | 'enabled' | 'disabled' }>((query) => store.listUsers(query), { keyword: '', orgId: '', status: '' })
const resource = useResource<UserInput>(() => ({ orgId: selectedOrgId.value || 'org-frontend', username: '', displayName: '', email: '', roleIds: ['role-viewer'], status: 'enabled' }), store.createUser, store.updateUser, table.reload)

function selectOrg(node: OrgNode): void { selectedOrgId.value = node.id; table.state.query.orgId = node.id; void table.search() }
function edit(user: User): void { resource.openEdit(user.id, { orgId: user.orgId, username: user.username, displayName: user.displayName, email: user.email, roleIds: [...user.roleIds], status: user.status }) }
function openRoles(user: User): void { roleUser.value = user; roleIds.value = [...user.roleIds] }
async function saveRoles(): Promise<void> { if (roleUser.value) { await store.assignRoles(roleUser.value.id, roleIds.value); roleUser.value = null; await table.reload() } }
function roleNames(user: User): string { return user.roleIds.map((id) => store.roleNameMap.get(id) ?? id).join('、') }
onMounted(() => void table.reload())
</script>

<template>
  <section class="grid gap-4 lg:grid-cols-[240px_1fr]">
    <aside class="rounded border p-3">
      <ul>
        <OrgTreeNode
          v-for="node in store.orgTree"
          :key="node.id"
          :node="node"
          :selected-id="selectedOrgId"
          @select="selectOrg"
        />
      </ul>
    </aside>

    <main class="rounded border p-3">
      <form class="mb-3 flex gap-2" @submit.prevent="table.search">
        <input v-model="table.state.query.keyword" class="border px-2 py-1" placeholder="搜索用户" />
        <select v-model="table.state.query.status" class="border px-2 py-1">
          <option value="">全部</option>
          <option value="enabled">启用</option>
          <option value="disabled">禁用</option>
        </select>
        <button type="submit">搜索</button>
        <button type="button" @click="resource.openCreate">新增</button>
      </form>

      <p v-if="table.error || resource.error" class="text-red-600">
        {{ table.error || resource.error }}
      </p>

      <table class="w-full text-sm">
        <tbody>
          <tr v-for="user in table.rows" :key="user.id" class="border-t">
            <td>{{ user.username }}</td>
            <td>{{ user.displayName }}</td>
            <td>{{ roleNames(user) }}</td>
            <td>{{ user.status }}</td>
            <td>
              <button @click="edit(user)">编辑</button>
              <button @click="openRoles(user)">角色</button>
              <button @click="store.toggleStatus(user.id).then(table.reload)">启停</button>
              <button @click="store.deleteUser(user.id).then(table.reload)">删除</button>
            </td>
          </tr>
        </tbody>
      </table>

      <div class="mt-3">
        <button @click="table.previousPage">上一页</button>
        <span>第 {{ table.state.page }} / {{ table.pageCount }} 页</span>
        <button @click="table.nextPage">下一页</button>
      </div>
    </main>

    <form
      v-if="resource.visible"
      class="fixed inset-10 bg-white p-4 shadow"
      @submit.prevent="resource.save"
    >
      <input v-model="resource.form.username" placeholder="用户名" />
      <input v-model="resource.form.displayName" placeholder="姓名" />
      <input v-model="resource.form.email" placeholder="邮箱" />
      <button>保存</button>
      <button type="button" @click="resource.close">取消</button>
    </form>

    <form
      v-if="roleUser"
      class="fixed inset-10 bg-white p-4 shadow"
      @submit.prevent="saveRoles"
    >
      <label v-for="role in store.roles" :key="role.id">
        <input v-model="roleIds" type="checkbox" :value="role.id" />{{ role.name }}
      </label>
      <button>保存角色</button>
    </form>
  </section>
</template>
