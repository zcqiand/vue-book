<!-- src/pages/BookList.vue -->
<script setup lang="ts">
import { ref } from 'vue'
import GenericList from '../components/GenericList.vue'

interface Book {
  id: number
  title: string
  author: string
}

const books = ref<Book[]>([
  { id: 1, title: 'Vue 3实战', author: '张三' },
])

function renderBook(item: Book, index: number): string {
  return `${index + 1}. 《${item.title}》- ${item.author}`
}

function renderBookIndex(item: Book, index: number): string {
  return `[${index}] ${item.title}`
}
</script>

<template>
  <div>
    <GenericList :items="books" :render-item="renderBook" />
    <GenericList :items="books" :render-item="renderBookIndex" />
  </div>
</template>
