<template>
  <div class="modal-demo">
    <button @click="show = !show" class="btn-toggle">
      {{ show ? '关闭模态框' : '打开模态框' }}
    </button>

    <!-- Transition: v-if 为 true 时添加 .fade-enter-from，
         1 frame 后切换到 .fade-enter-active，
         进入动画完成后移除 -->
    <Transition name="fade">
      <div v-if="show" class="modal-overlay">
        <div class="modal">
          <h3>模态框标题</h3>
          <p>这是模态框的内容，带有淡入淡出动画。</p>
          <button @click="show = false" class="btn-close">关闭</button>
        </div>
      </div>
    </Transition>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'

const show = ref<boolean>(false)
</script>
