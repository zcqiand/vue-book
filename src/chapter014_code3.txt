<template>
  <input ref="inputEl" type="text" />
  <button @click="focusInput">聚焦</button>
</template>

<script setup lang="ts">
import { ref, onMounted } from 'vue'

// 用 ref() 声明一个与模板 ref 同名的变量
const inputEl = ref<HTMLInputElement | null>(null)

function focusInput() {
  // 在 setup 中访问 ref 模板引用，需要 .value
  inputEl.value?.focus()
}

// 或者在 onMounted 中自动聚焦
onMounted(() => {
  inputEl.value?.focus()
})
</script>
