<template>
  <div class="product-page">
    <h2>商品详情</h2>

    <Suspense>
      <template #default>
        <ErrorBoundary>
          <ProductCard :productId="productId" />
        </ErrorBoundary>
      </template>

      <template #fallback>
        <div class="product-skeleton">
          <div class="skeleton-img"></div>
          <div class="skeleton-text long"></div>
          <div class="skeleton-text medium"></div>
          <div class="skeleton-text short"></div>
        </div>
      </template>

      <template #error="{ error }">
        <div class="product-error">
          <div class="error-icon">!</div>
          <h3>商品加载失败</h3>
          <p>{{ error.message }}</p>
          <button class="btn-retry" @click="retryProduct">重新加载商品</button>
        </div>
      </template>
    </Suspense>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import { defineAsyncComponent } from 'vue'
import ErrorBoundary from '../components/ErrorBoundary.vue'

const productId = ref(42)

// 异步加载商品卡片组件（模拟动态 import）
const ProductCard = defineAsyncComponent(() => import('../components/ProductCard.vue'))

// 重试函数：强制刷新当前页面，触发 Suspense 重新加载
function retryProduct(): void {
  // 刷新当前路由（或重新设置 key 触发组件重新渲染）
  location.reload()
}
</script>
