<!-- src/components/ThemedButton.vue -->
<template>
  <div class="themed-button-container">
    <!-- 显示当前主题名称 -->
    <span class="theme-label">当前主题：{{ theme }}</span>

    <!-- 点击按钮切换主题 -->
    <button
      class="theme-toggle-btn"
      @click="switchTheme"
    >
      切换主题
    </button>
  </div>
</template>

<script setup lang="ts">
import { inject } from 'vue'

// inject 第一个参数：与 provide 的 key 对应（字符串 Symbol）
// inject 第二个参数（可选）：默认值，这里不设置因为 theme 必定被提供
const theme = inject<import('vue').Ref<'light' | 'dark'>>('theme')
const switchTheme = inject<() => void>('switchTheme')

// 如果祖先未提供，inject 返回 undefined（但 ThemeProvider 必定提供）
if (!theme || !switchTheme) {
  throw new Error('ThemedButton 必须用在 ThemeProvider 内部')
}
</script>

.themed-button-container {
