<script setup lang="ts">
import { ref } from 'vue'

const hasError = ref<boolean>(false)
const errorMessage = ref<string>('')

onErrorCaptured((err, _instance, info) => {
  hasError.value = true
  errorMessage.value = err instanceof Error ? err.message : String(err)
  console.error('捕获到子组件错误:', err, '堆栈信息:', info)
  return false // 阻止错误继续向上冒泡
})

function retry(): void {
  hasError.value = false
  errorMessage.value = ''
}
</script>

<template>
  <div class="error-boundary">
    <!-- 正常内容 -->
    <slot v-if="!hasError" />

    <!-- 降级 UI：错误态 -->
    <div v-else class="fallback-ui">
      <div class="error-icon">⚠️</div>
      <h3>页面遇到了一些问题</h3>
      <p>{{ errorMessage }}</p>
      <button @click="retry" class="retry-btn">重新加载</button>
    </div>
  </div>
</template>

<style scoped>
.error-boundary { min-height: 100px; }
.fallback-ui {
  padding: 32px;
  text-align: center;
  background: #fef2f2;
  border: 1px solid #fecaca;
  border-radius: 8px;
}
.error-icon { font-size: 32px; margin-bottom: 12px; }
.retry-btn {
  margin-top: 12px;
  padding: 8px 20px;
  background: #dc2626;
  color: white;
  border: none;
  border-radius: 4px;
  cursor: pointer;
}
</style>
