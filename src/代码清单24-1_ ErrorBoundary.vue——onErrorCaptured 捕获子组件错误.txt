<!-- src/components/ErrorBoundary.vue -->
<template>
  <div class="error-boundary">
    <!-- 正常态：渲染子组件插槽内容 -->
    <slot v-if="!hasError" />

    <!-- 降级态：错误时显示友好提示 + 重试按钮 -->
    <div v-else class="fallback-ui">
      <div class="error-icon">!</div>
      <h3>页面出现了一些问题</h3>
      <p>不用担心，这只是局部内容加载失败。</p>
      <button class="btn-retry" @click="handleRetry">重新加载</button>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, onErrorCaptured } from 'vue'

// 错误态标志
const hasError = ref(false)

// onErrorCaptured：捕获从子组件树上抛上来的错误
// err: 错误对象 instance: 抛出错误的组件实例 info: 额外调试信息
onErrorCaptured((err: Error, instance: object | null, info: string) => {
  console.error('[ErrorBoundary] 捕获到子组件错误:', err.message, info)

  // 标记错误态，渲染降级 UI
  hasError.value = true

  // 返回 false：阻止错误继续向父组件冒泡
  return false
})

// 重试按钮处理函数：清除错误态，让 Suspense 重新加载
function handleRetry(): void {
  hasError.value = false
}
</script>
