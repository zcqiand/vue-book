<!-- src/pages/UserProfile.vue -->
<template>
  <div class="user-profile-page">
    <h2>用户资料</h2>

    <!-- Suspense 包裹异步组件，监听加载状态 -->
    <Suspense>
      <!-- #default：正常加载完成的组件 -->
      <template #default>
        <ErrorBoundary>
          <UserCard :userId="userId" />
        </ErrorBoundary>
      </template>

      <!-- #fallback：加载中的占位 UI -->
      <template #fallback>
        <div class="loading-skeleton">
          <div class="skeleton-avatar"></div>
          <div class="skeleton-lines">
            <div class="skeleton-line"></div>
            <div class="skeleton-line short"></div>
          </div>
        </div>
      </template>

      <!-- #error：异步组件加载失败时的错误态 -->
      <template #error="{ error }">
        <div class="error-state">
          <div class="error-icon">!</div>
          <h3>加载失败</h3>
          <p>{{ error.message || '网络请求异常，请检查网络后重试。' }}</p>
          <button class="btn-retry" @click="retryLoad">重新加载</button>
        </div>
      </template>
    </Suspense>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import { defineAsyncComponent } from 'vue'
import ErrorBoundary from '../components/ErrorBoundary.vue'

// 模拟用户 ID（可从路由参数获取）
const userId = ref(1)

// defineAsyncComponent：异步加载 UserCard，模拟网络请求
const UserCard = defineAsyncComponent(() => import('../components/UserCard.vue'))

// 重试加载：Vue 会重新执行异步导入
function retryLoad(): void {
  // 强制组件重新渲染即可重新触发 Suspense 加载
  location.reload()
}
</script>
