<script setup lang="ts">
import { computed, onMounted, reactive, ref, shallowRef, type Ref } from 'vue'
import { useUserStore, type AuditLog, type AuditSnapshot } from '@/stores/user'

function useVirtualList<T>(items: Ref<T[]>, itemHeight: number, containerHeight: number) {
  const scrollTop = ref(0)
  const start = computed(() => Math.max(0, Math.floor(scrollTop.value / itemHeight) - 4))
  const end = computed(() => Math.min(items.value.length, start.value + Math.ceil(containerHeight / itemHeight) + 8))
  const totalHeight = computed(() => items.value.length * itemHeight)
  const rows = computed(() => items.value.slice(start.value, end.value).map((item, offset) => ({ item, index: start.value + offset, top: (start.value + offset) * itemHeight })))
  function onScroll(event: Event): void { scrollTop.value = (event.currentTarget as HTMLElement).scrollTop }
  function reset(): void { scrollTop.value = 0 }
  return { rows, totalHeight, onScroll, reset }
}

const store = useUserStore()
const logs = shallowRef<AuditLog[]>([])
const selectedLog = ref<AuditLog | null>(null)
const error = ref('')
const loading = ref(false)
const pager = reactive({ page: 1, pageSize: 50, total: 0 })
const pageCount = computed(() => Math.max(1, Math.ceil(pager.total / pager.pageSize)))
const list = useVirtualList(logs, 76, 460)
const drawerVisible = computed(() => selectedLog.value !== null)

async function loadLogs(page = pager.page): Promise<void> {
  loading.value = true
  error.value = ''
  try {
    const nextPage = Math.min(Math.max(1, page), pageCount.value)
    const result = await store.fetchAuditLogs({ page: nextPage, pageSize: pager.pageSize })
    logs.value = result.items
    pager.total = result.total
    pager.page = nextPage
    selectedLog.value = null
    list.reset()
  } catch (caught) {
    error.value = caught instanceof Error ? caught.message : '加载审计日志失败'
  } finally {
    loading.value = false
  }
}

async function previousPage(): Promise<void> {
  if (pager.page > 1) await loadLogs(pager.page - 1)
}

async function nextPage(): Promise<void> {
  if (pager.page < pageCount.value) await loadLogs(pager.page + 1)
}

function actionText(action: AuditLog['action']): string {
  return ({ create: '创建用户', update: '更新用户', delete: '删除用户', assignRoles: '分配角色', toggleStatus: '启用/禁用' })[action]
}
function formatJson(value: AuditSnapshot | null): string { return value ? JSON.stringify(value, null, 2) : '无' }
function updateFirstActor(): void { if (logs.value[0]) { const [first, ...rest] = logs.value; logs.value = [{ ...first, actor: 'auditor@tenant-a.local' }, ...rest] } }
onMounted(() => void loadLogs(1))
</script>

<template>
  <section class="rounded border p-4">
    <div class="mb-3 flex justify-between"><h1>审计日志</h1><div><button @click="updateFirstActor">整体替换示例</button><button @click="loadLogs(pager.page)">刷新</button></div></div>
    <p class="mb-2 text-sm text-slate-500">服务端分页：第 {{ pager.page }} / {{ pageCount }} 页，共 {{ pager.total }} 条；当前页 {{ logs.length }} 条，虚拟滚动只优化当前页渲染。</p>
    <p v-if="error" class="text-red-600">{{ error }}</p>
    <p v-if="loading" class="text-slate-500">加载中...</p>
    <div class="relative overflow-y-auto border" style="height: 460px" @scroll="list.onScroll">
      <div :style="{ height: `${list.totalHeight.value}px`, position: 'relative' }">
        <button v-for="row in list.rows.value" :key="row.item.id" type="button" class="absolute left-0 right-0 grid grid-cols-4 border-b px-3 py-2 text-left" :style="{ transform: `translateY(${row.top}px)`, height: '76px' }" @click="selectedLog = row.item">
          <span>{{ actionText(row.item.action) }}</span><span>{{ row.item.targetName }}</span><span>{{ row.item.actor }}</span><span>{{ new Date(row.item.createdAt).toLocaleString() }}</span>
        </button>
      </div>
    </div>
    <div class="mt-3 flex items-center gap-3"><button type="button" :disabled="pager.page <= 1" @click="previousPage">上一页</button><button type="button" :disabled="pager.page >= pageCount" @click="nextPage">下一页</button></div>
    <aside v-if="drawerVisible && selectedLog" class="fixed inset-y-0 right-0 w-full max-w-lg overflow-auto bg-white p-4 shadow">
      <button @click="selectedLog = null">关闭</button><h2>日志详情</h2>
      <p>操作人：{{ selectedLog.actor }}</p><p>时间：{{ new Date(selectedLog.createdAt).toLocaleString() }}</p><p>操作：{{ actionText(selectedLog.action) }}</p>
      <h3>变更前</h3><pre>{{ formatJson(selectedLog.before) }}</pre><h3>变更后</h3><pre>{{ formatJson(selectedLog.after) }}</pre>
    </aside>
  </section>
</template>
