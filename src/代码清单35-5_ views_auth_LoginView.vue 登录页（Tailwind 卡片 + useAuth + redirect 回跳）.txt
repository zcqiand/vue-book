<script setup lang="ts">
import { ref } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import { useAuth } from '@/composables/useAuth'

const { login, loading, error } = useAuth()
const route = useRoute()
const router = useRouter()

// 表单数据用 ref 维护，绑定到 input 的 v-model（g-032）
const username = ref<string>('')
const password = ref<string>('')

// 表单级校验错误（与 useAuth 的 error 区分：那个是接口错误，这个是本地校验）
const validationError = ref<string>('')

async function handleSubmit(): Promise<void> {
  // 简单非空校验：登录接口前先把空输入拦下，省一次无谓的网络请求
  validationError.value = ''
  if (!username.value.trim() || !password.value.trim()) {
    validationError.value = '用户名与密码不能为空'
    return
  }

  // ❶ 调 useAuth 的 login，loading 与 error 由 Composable 内部维护
  const ok = await login(username.value.trim(), password.value)
  if (!ok) {
    return
  }

  // ❷ 登录成功后读 query.redirect 回跳目标路由
  // 用 replace 而非 push：避免用户点返回键又回到登录页，造成「我明明登录了」的困惑
  // 无 redirect 参数时默认跳工作台
  const redirect = route.query.redirect
  const target = typeof redirect === 'string' && redirect.startsWith('/') ? redirect : '/dashboard'
  router.replace(target)
}
</script>

<template>
  <!-- ❸ min-h-screen 居中卡片：用 flex 把卡片在视口中央对齐 -->
  <div class="min-h-screen flex items-center justify-center bg-gray-50 px-4">
    <div class="w-full max-w-md bg-white rounded-lg shadow-md p-8">
      <h1 class="text-2xl font-bold text-gray-800 mb-6 text-center">实验室管理系统</h1>

      <!-- 表单用 @submit.prevent 阻止默认提交刷新页面，改走 handleSubmit -->
      <form class="flex flex-col gap-4" @submit.prevent="handleSubmit">
        <div class="flex flex-col gap-1">
          <label class="text-sm text-gray-600" for="username">用户名</label>
          <input
            id="username"
            v-model.trim="username"
            type="text"
            autocomplete="username"
            class="px-3 py-2 border border-gray-300 rounded focus:outline-none focus:border-blue-500"
            placeholder="请输入用户名"
          />
        </div>

        <div class="flex flex-col gap-1">
          <label class="text-sm text-gray-600" for="password">密码</label>
          <input
            id="password"
            v-model.trim="password"
            type="password"
            autocomplete="current-password"
            class="px-3 py-2 border border-gray-300 rounded focus:outline-none focus:border-blue-500"
            placeholder="请输入密码"
          />
        </div>

        <!-- 本地校验错误提示 -->
        <p v-if="validationError" class="text-sm text-red-600">{{ validationError }}</p>
        <!-- 接口错误提示 -->
        <p v-if="error" class="text-sm text-red-600">{{ error }}</p>

        <button
          type="submit"
          :disabled="loading"
          class="px-4 py-2 bg-blue-600 text-white rounded hover:bg-blue-700 disabled:bg-gray-400 disabled:cursor-not-allowed transition-colors"
        >
          {{ loading ? '登录中...' : '登录' }}
        </button>
      </form>
    </div>
  </div>
</template>
