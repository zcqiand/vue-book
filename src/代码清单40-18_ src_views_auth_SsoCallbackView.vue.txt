<script setup lang="ts">
import { onMounted, ref } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import { useSso } from '@/composables/useSso'

const route = useRoute()
const router = useRouter()
const { handleSsoCallback } = useSso()
const errorMessage = ref('')

onMounted(async () => {
  try {
    await handleSsoCallback(route, router)
  } catch (error) {
    errorMessage.value = error instanceof Error ? error.message : 'SSO 登录失败'
  }
})
</script>

<template>
  <section class="rounded-xl border border-slate-200 bg-white p-6 shadow-sm">
    <h2 class="text-lg font-semibold">正在处理 SSO 回调</h2>
    <p v-if="errorMessage.length === 0" class="mt-2 text-sm text-slate-600">正在换取本系统 token，请稍候。</p>
    <p v-else class="mt-2 rounded-lg bg-red-50 p-3 text-sm text-red-700">{{ errorMessage }}</p>
  </section>
</template>
