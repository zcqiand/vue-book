<script setup lang="ts">
import { computed } from 'vue'
import { RouterLink } from 'vue-router'
import { useAuthStore } from '@/stores/auth'

// ❶ 菜单配置：每项声明可见性条件（roles 或 permission），由 computed 过滤
// 配置而非散落模板，是为了后续从后端拉菜单、做配置化菜单系统时改一处即可
interface MenuItem {
  label: string
  to: string
  /** 任一角色匹配即显示（OR 语义） */
  roles?: string[]
  /** 拥有该权限即显示 */
  permission?: string
}

const menuItems: MenuItem[] = [
  { label: '工作台', to: '/dashboard' },
  { label: '检测项目', to: '/inspection' },
  { label: '样品管理', to: '/sample' },
  { label: '报告管理', to: '/report' },
  // 演示角色过滤：只有 admin 能看到用户管理入口
  { label: '用户管理', to: '/admin/users', roles: ['admin'] },
  // 演示权限过滤：拥有 log:delete 权限的人才能看到删除日志入口
  { label: '删除日志', to: '/admin/logs', permission: 'log:delete' }
]

const authStore = useAuthStore()

// ❷ computed 派生可见菜单：roles 与 permission 任一通过即显示
const visibleMenus = computed<MenuItem[]>(() => {
  return menuItems.filter((item) => {
    if (!item.roles && !item.permission) {
      return true
    }
    if (item.roles && item.roles.some((r) => authStore.hasRole(r))) {
      return true
    }
    if (item.permission && authStore.hasPermission(item.permission)) {
      return true
    }
    return false
  })
})
</script>

<template>
  <aside class="w-56 bg-gray-50 border-r border-gray-200 py-4">
    <nav class="flex flex-col gap-1 px-2">
      <!-- ❸ v-for 渲染 visibleMenus，每项 RouterLink 沿用 34.10 的 active-class 高亮 -->
      <RouterLink
        v-for="item in visibleMenus"
        :key="item.to"
        :to="item.to"
        class="px-3 py-2 rounded text-gray-700 hover:bg-gray-100"
        active-class="bg-blue-50 text-blue-700 font-medium"
      >
        {{ item.label }}
      </RouterLink>
    </nav>
  </aside>
</template>

<style scoped>
/* 样式全部走 Tailwind，scoped 块留作扩展 */
</style>
