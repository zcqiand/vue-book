<!-- src/views/UserBadge.vue -->
<script setup lang="ts">
import { storeToRefs } from 'pinia'
import { useUserStore } from '@/stores/user'

const userStore = useUserStore()

// storeToRefs 将 store 属性转为 ref，解构后仍保持响应式
const { token, userInfo } = storeToRefs(userStore)

// actions 不需要 storeToRefs，直接从 store 实例获取即可
const { login, logout } = userStore

// 错误示范（不要这样写）：
// const { token, userInfo } = userStore
// 以上写法解出的 token 和 userInfo 是普通值，不是响应式 ref，
// 组件无法感知 store 状态变化，会导致 UI 不更新的 bug
</script>

<template>
  <div class="user-badge">
    <template v-if="token">
      <!-- token 是 ref，解构后通过 .value 读取；模板中自动解包无需 .value -->
      <img
        v-if="userInfo?.avatar"
        :src="userInfo.avatar"
        :alt="userInfo.username"
        class="avatar"
      />
      <span>{{ userInfo?.username }}</span>
    </template>
    <template v-else>
      <span class="anonymous">未登录</span>
    </template>
  </div>
</template>

<style scoped>
.user-badge {
  display: flex;
  align-items: center;
  gap: 8px;
}
.avatar {
  width: 32px;
  height: 32px;
  border-radius: 50%;
}
</style>
