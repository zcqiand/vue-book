<template>
  <div class="contact-manager">
    <h2>联系方式管理</h2>

    <div class="contact-list">
      <!-- v-for 遍历 contacts 数组，每项渲染一行 -->
      <div v-for="(contact, index) in contacts" :key="index" class="contact-row">
        <!-- 联系方式类型选择 -->
        <select v-model="contact.type" class="type-select">
          <option value="phone">电话</option>
          <option value="email">邮箱</option>
          <option value="wechat">微信</option>
          <option value="other">其他</option>
        </select>

        <!-- 联系方式值输入 -->
        <input
          v-model="contact.value"
          type="text"
          :placeholder="getPlaceholder(contact.type)"
          class="value-input"
        />

        <!-- 实时错误提示 -->
        <span v-if="contactErrors[index]" class="error-msg">{{ contactErrors[index] }}</span>

        <!-- 删除按钮 -->
        <button
          type="button"
          class="btn-remove"
          @click="removeContact(index)"
          :disabled="contacts.length <= 1"
        >
          删除
        </button>
      </div>
    </div>

    <!-- 添加新联系方式 -->
    <button type="button" class="btn-add" @click="addContact">
      + 添加联系方式
    </button>

    <!-- 调试信息（开发时可见） -->
    <details class="debug-info">
      <summary>调试：查看当前数据</summary>
      <pre>{{ JSON.stringify(contacts, null, 2) }}</pre>
    </details>
  </div>
</template>

<script setup lang="ts">
import { reactive, computed } from 'vue'

// 联系方式项的类型
interface ContactItem {
  type: string
  value: string
}

// reactive 数组管理多个联系方式
const contacts = reactive<ContactItem[]>([
  { type: 'phone', value: '' },
  { type: 'email', value: '' },
])

// 邮箱格式验证
function validateEmail(value: string): string | null {
  if (!value) return null // 空值时跳过校验（可用 required 补充）
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/
  return emailRegex.test(value) ? null : '邮箱格式不正确'
}

// 根据类型返回 placeholder
function getPlaceholder(type: string): string {
  const placeholders: Record<string, string> = {
    phone: '13800138000',
    email: 'example@domain.com',
    wechat: 'your-wechat-id',
    other: '其他联系方式',
  }
  return placeholders[type] ?? '请输入'
}

// 添加一项空联系方式
function addContact(): void {
  contacts.push({ type: 'email', value: '' })
}

// 删除指定索引的项
function removeContact(index: number): void {
  if (contacts.length > 1) {
    contacts.splice(index, 1)
  }
}

// 每项的错误信息（computed 响应式数组，每次 contacts 变化时自动重算）
const contactErrors = computed<(string | null)[]>(() =>
  contacts.map((contact) => {
    if (contact.type === 'email' && contact.value) {
      return validateEmail(contact.value)
    }
    return null
  })
)
</script>
