<template>
  <div class="deferred-teleport-demo">
    <p class="hint">点击按钮后，defer Teleport 的内容会被传送到下方的目标容器中。</p>

    <button @click="show = !show" class="btn-toggle">
      {{ show ? '移除内容' : '传送内容（defer）' }}
    </button>

    <!-- defer Teleport：等到 #later-target 渲染后再执行传送 -->
    <!-- 如果不用 defer，#later-target 尚未渲染则传送失败，内容消失 -->
    <Teleport defer to="#later-target">
      <div v-if="show" class="deferred-content">
        <span class="content-badge">这是 defer Teleport 传送来的内容</span>
      </div>
    </Teleport>

    <!-- 目标容器：Teleport 之后才渲染 -->
    <div id="later-target" class="target-container">
      <span v-if="!show" class="placeholder">目标容器（等待内容传送）</span>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'

const show = ref<boolean>(false)
</script>
