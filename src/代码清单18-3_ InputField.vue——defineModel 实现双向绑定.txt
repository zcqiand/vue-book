<!-- src/components/InputField.vue -->
<template>
  <div class="input-field">
    <!-- label 在上 -->
    <label v-if="label" :for="inputId" class="field-label">{{ label }}</label>

    <!-- 输入框在中 -->
    <textarea
      v-if="type === 'textarea'"
      :id="inputId"
      v-model="modelValue"
      :placeholder="placeholder"
      class="field-input field-textarea"
    />
    <input
      v-else
      :id="inputId"
      v-model="modelValue"
      :type="type"
      :placeholder="placeholder"
      :autocomplete="autocomplete"
      class="field-input"
    />

    <!-- 错误信息红色显示在下 -->
    <span v-if="localError" class="field-error">{{ localError }}</span>
  </div>
</template>

<script setup lang="ts">
import { computed } from 'vue'
import { type validatorFn } from '../composables/useValidation'

// defineModel 自动处理 v-model 的 props 和 emits
// 无需手动 defineProps + defineEmits，编译器自动展开为 modelValue prop + update:modelValue emit
const modelValue = defineModel<string>({ required: true })

const props = defineProps<{
  label?: string
  type?: 'text' | 'email' | 'password' | 'textarea'
  placeholder?: string
  rules?: validatorFn[]
  autocomplete?: string
}>()

// 生成唯一 id，用于 label 的 for 属性（Vue 3.5+ 推荐用 useId()，此处兼容实现）
const inputId = `input-${Math.random().toString(36).slice(2, 9)}`

// 本地错误状态：computed 来自 rules 校验结果
const localError = computed<string | null>(() => {
  if (!props.rules || props.rules.length === 0) {
    return null
  }
  for (const rule of props.rules) {
    const result = rule(modelValue.value)
    if (result !== null) {
      return result
    }
  }
  return null
})
</script>
