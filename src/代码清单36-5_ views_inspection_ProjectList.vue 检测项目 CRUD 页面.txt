<script setup lang="ts">
import { ref, onMounted } from 'vue'
import { useResource } from '@/composables/useResource'
import { useTable } from '@/composables/useTable'
import {
  listProjects,
  createProject,
  updateProject,
  removeProject,
  type Project,
  type ProjectSearch,
  type ProjectPayload
} from '@/api/inspection'
import { useInspectionStore } from '@/stores/inspection'
import ProjectForm from '@/components/ProjectForm.vue'

const inspectionStore = useInspectionStore()
const resource = useResource<Project, ProjectPayload, ProjectSearch>({
  list: listProjects,
  create: createProject,
  update: updateProject,
  remove: removeProject
})
const table = useTable<ProjectSearch>({ initialSearch: { name: '', status: undefined } })

table.bindReload(() => resource.list(table.queryParams.value))

const formVisible = ref(false)
const editingId = ref<Project['id'] | null>(null)
const formData = ref<{ name: string; client: string; status: Project['status'] }>({
  name: '',
  client: '',
  status: 'pending'
})

onMounted(() => {
  resource.list(table.queryParams.value)
  inspectionStore.fetchStatuses()
})

function statusLabel(value: Project['status']): string {
  return inspectionStore.statusMap.get(value) ?? value
}

function totalPages(): number {
  return Math.max(1, Math.ceil(resource.total.value / table.pageSize.value))
}

function openCreate(): void {
  editingId.value = null
  formData.value = { name: '', client: '', status: 'pending' }
  formVisible.value = true
}

function openEdit(project: Project): void {
  editingId.value = project.id
  formData.value = { name: project.name, client: project.client, status: project.status }
  formVisible.value = true
}

async function handleSubmit(payload: ProjectPayload): Promise<void> {
  const ok = editingId.value === null
    ? await resource.create(payload)
    : await resource.update(editingId.value, payload)
  if (ok) formVisible.value = false
}

function handleCancel(): void {
  formVisible.value = false
}

async function handleRemove(project: Project): Promise<void> {
  const confirmed = window.confirm(`确认删除项目「${project.name}」吗？此操作不可撤销。`)
  if (confirmed) await resource.remove(project.id)
}

function handleSearch(): void {
  table.page.value = 1
  table.reload()
}
</script>

<template>
  <div>
    <div v-if="resource.error.value">{{ resource.error.value }}</div>
    <form @submit.prevent="handleSearch">
      <input v-model="table.search.name" type="text" placeholder="按名称模糊搜索" />
      <select v-model="table.search.status">
        <option :value="undefined">全部</option>
        <option v-for="option in inspectionStore.statusOptions" :key="option.value" :value="option.value">{{ option.label }}</option>
      </select>
      <button type="submit">搜索</button>
      <button type="button" @click="table.resetSearch()">重置</button>
      <button type="button" @click="openCreate">新增项目</button>
    </form>

    <table>
      <thead>
        <tr><th>项目名称</th><th>委托单位</th><th>状态</th><th>创建时间</th><th>操作</th></tr>
      </thead>
      <tbody>
        <tr v-if="resource.loading.value"><td colspan="5">加载中...</td></tr>
        <tr v-else-if="resource.items.value.length === 0"><td colspan="5">暂无数据</td></tr>
        <tr v-for="project in resource.items.value" :key="project.id">
          <td>{{ project.name }}</td>
          <td>{{ project.client }}</td>
          <td>{{ statusLabel(project.status) }}</td>
          <td>{{ project.createdAt }}</td>
          <td>
            <button type="button" @click="openEdit(project)">编辑</button>
            <button type="button" @click="handleRemove(project)">删除</button>
          </td>
        </tr>
      </tbody>
    </table>

    <button type="button" :disabled="table.page.value <= 1" @click="table.changePage(table.page.value - 1)">上一页</button>
    <span>{{ table.page.value }} / {{ totalPages() }}</span>
    <button type="button" :disabled="table.page.value >= totalPages()" @click="table.changePage(table.page.value + 1)">下一页</button>

    <ProjectForm v-model="formData" :visible="formVisible" @submit="handleSubmit" @cancel="handleCancel" />
  </div>
</template>
