<script setup lang="ts">
import { useRouter } from 'vue-router'

const router = useRouter()
const pageTitle = '用户列表'

interface User {
  id: number
  name: string
}

const users: User[] = [
  { id: 1, name: '张三' },
  { id: 2, name: '李四' },
  { id: 3, name: '王五' },
]

function goToDetail(id: number): void {
  router.push({ name: 'user-detail', params: { id } })
}
</script>

<template>
  <div>
    <h1 class="text-2xl font-bold text-gray-800 mb-4">{{ pageTitle }}</h1>
    <ul class="space-y-2">
      <li
        v-for="user in users"
        :key="user.id"
        class="p-3 bg-white rounded border border-gray-200 cursor-pointer hover:bg-indigo-50 transition-colors"
        @click="goToDetail(user.id)"
      >
        {{ user.id }} - {{ user.name }}
      </li>
    </ul>
  </div>
</template>
