<script setup lang="ts">
// withDefaults 接收两个参数：defineProps 的返回结果（这里直接传入调用）和一个默认值对象
withDefaults(
  defineProps<{
    label: string
    variant?: 'default' | 'success' | 'danger'
    size?: 'sm' | 'md' | 'lg'
  }>(),
  {
    variant: 'default',
    size: 'md',
  },
)
</script>

<template>
  <span :class="['badge', `badge-${variant}`]">{{ label }}</span>
</template>
