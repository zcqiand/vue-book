<script setup lang="ts">
import { ref, reactive } from 'vue'

interface User {
  name: string
  age: number
}

// ===== 方案A：reactive 适合「改属性」，但不适合「整体替换」 =====
const reactiveUser = reactive<User>({ name: '张三', age: 28 })

function patchReactive(): void {
  // 改属性：reactive 的强项，直接赋值即可
  reactiveUser.name = '李四'
  reactiveUser.age = 30
}

// ❌ 错误示范：下面这样整体重新赋值会丢掉响应式（reactive 返回的是代理对象本身，
//    重新赋值只是让变量名指向了新对象，模板里绑定的还是旧代理）
// function replaceReactive() {
//   reactiveUser = { name: '王五', age: 35 } // TS 报错：不能给 const 重新赋值
// }

// ===== 方案B：ref 包对象，既能改属性，又能整体替换 =====
const refUser = ref<User>({ name: '赵六', age: 40 })

function patchRef(): void {
  // 改属性：通过 .value 访问内部对象，再改属性
  refUser.value.name = '钱七'
  refUser.value.age = 25
}

function replaceRef(): void {
  // 整体替换：把 .value 指向全新对象，模板自动响应新数据
  // 这是 ref 相对 reactive 的核心优势之一
  refUser.value = { name: '孙八', age: 50 }
}
</script>

<template>
  <div>
    <h3>reactive 用户</h3>
    <p>{{ reactiveUser.name }} / {{ reactiveUser.age }} 岁</p>
    <button @click="patchReactive">改属性（生效）</button>

    <h3>ref 用户</h3>
    <p>{{ refUser.name }} / {{ refUser.age }} 岁</p>
    <button @click="patchRef">改属性</button>
    <button @click="replaceRef">整体替换（仅 ref 能干净替换）</button>
  </div>
</template>
