<!-- src/components/ThemeProvider.vue -->
<template>
  <div class="theme-provider" :class="theme">
    <slot :theme="theme" :switch-theme="switchTheme" />
  </div>
</template>

<script setup lang="ts">
import { ref, provide } from 'vue'
import { themeKey, switchThemeKey } from '../symbols'

type Theme = 'light' | 'dark'

const theme = ref<Theme>('light')

function switchTheme(): void {
  theme.value = theme.value === 'light' ? 'dark' : 'light'
}

// 使用 InjectionKey 提供，类型自动携带
provide(themeKey, theme)
provide(switchThemeKey, switchTheme)
</script>
